   111→function planBadge(plan: string): string {
   112→  const map: Record<string, string> = {
   113→    free: "bg-slate-100 text-slate-700 border-slate-200",
   114→    starter: "bg-sky-50 text-sky-700 border-sky-200",
   115→    pro: "bg-emerald-50 text-emerald-700 border-emerald-200",
   116→    agency: "bg-violet-50 text-violet-700 border-violet-200",
   117→  };
   118→  return map[plan] ?? "bg-slate-100 text-slate-700 border-slate-200";
   119→}

   121→function roleBadge(role: string): string {
   122→  return role === "admin"
   123→    ? "bg-emerald-50 text-emerald-700 border-emerald-200"
   124→    : "bg-slate-100 text-slate-700 border-slate-200";
   125→}

   127→function scoreColor(v: number): string {
   128→  return v >= 80 ? "text-emerald-600" : v >= 60 ? "text-amber-600" : v >= 40 ? "text-orange-600" : "text-red-600";
   129→}

   131→function formatDate(iso: string): string {
   132→  if (!iso) return "—";
   133→  const d = new Date(iso);
   134→  if (isNaN(d.getTime())) return "—";
   135→  return d.toLocaleString(undefined, {
   136→    month: "short",
   137→    day: "numeric",
   138→    hour: "2-digit",
   139→    minute: "2-digit",
   140→  });
   141→}
   142→
   143→function formatDateLong(iso: string): string {
   144→  if (!iso) return "—";
   145→  const d = new Date(iso);
   146→  if (isNaN(d.getTime())) return "—";
   147→  return d.toLocaleDateString(undefined, {
   148→    year: "numeric",
   149→    month: "short",
   150→    day: "numeric",
   151→  });
   152→}
   153→
   154→function initials(name: string | null, email: string): string {
   155→  if (name) {
   156→    const parts = name.trim().split(/\s+/);
   157→    return (parts[0]?.[0] ?? "") + (parts[1]?.[0] ?? parts[0]?.[1] ?? "");
   158→  }
   159→  return email.slice(0, 2);
   160→}

   591→  return (

   612→function AddUserDialog({
   613→  open,
   614→  onOpenChange,
   615→  onCreated,
   616→}: {
   617→  open: boolean;
   618→  onOpenChange: (o: boolean) => void;
   619→  onCreated: () => void;
   620→}) {
   621→  const [email, setEmail] = React.useState("");
   622→  const [name, setName] = React.useState("");
   623→  const [role, setRole] = React.useState<string>("user");
   624→  const [plan, setPlan] = React.useState<string>("free");
   625→  const [saving, setSaving] = React.useState(false);
   626→
   627→  async function submit() {
   628→    if (!email.trim()) {
   629→      toast.error("Email is required");
   630→      return;
   631→    }
   632→    setSaving(true);
   633→    try {
   634→      const res = await fetch("/api/admin/users", {
   635→        method: "POST",
   636→        headers: { "Content-Type": "application/json" },
   637→        body: JSON.stringify({
   638→          email: email.trim(),
   639→          name: name.trim() || undefined,
   640→          role,
   641→          plan,
   642→        }),
   643→      });
   644→      if (!res.ok) {
   645→        const err = (await res.json().catch(() => ({}))) as { error?: string };
   646→        throw new Error(err.error ?? "create failed");
   647→      }
   648→      toast.success("User created");
   649→      setEmail("");
   650→      setName("");
   651→      setRole("user");
   652→      setPlan("free");
   653→      onOpenChange(false);
   654→      onCreated();
   655→    } catch (e) {
   656→      console.error(e);
   657→      toast.error(e instanceof Error ? e.message : "Failed to create user");
   658→    } finally {
   659→      setSaving(false);
   660→    }
   661→  }
   662→
   663→  return (
   664→    <Dialog open={open} onOpenChange={onOpenChange}>
   665→      <DialogContent className="sm:max-w-md">
   666→        <DialogHeader>
   667→          <DialogTitle>Add new user</DialogTitle>
   668→        </DialogHeader>
   669→        <div className="grid gap-3">
   670→          <div className="grid gap-1.5">
   671→            <Label htmlFor="au-email">Email</Label>
   672→            <Input
   673→              id="au-email"
   674→              type="email"
   675→              placeholder="user@example.com"
   676→              value={email}
   677→              onChange={(e) => setEmail(e.target.value)}
   678→            />
   679→          </div>
   680→          <div className="grid gap-1.5">
   681→            <Label htmlFor="au-name">Name (optional)</Label>
   682→            <Input
   683→              id="au-name"
   684→              placeholder="Jane Doe"
   685→              value={name}
   686→              onChange={(e) => setName(e.target.value)}
   687→            />
   688→          </div>
   689→          <div className="grid grid-cols-2 gap-3">
   690→            <div className="grid gap-1.5">
   691→              <Label>Role</Label>
   692→              <Select value={role} onValueChange={setRole}>
   693→                <SelectTrigger className="w-full">
   694→                  <SelectValue />
   695→                </SelectTrigger>
   696→                <SelectContent>
   697→                  {ROLES.map((r) => (
   698→                    <SelectItem key={r} value={r} className="capitalize">
   699→                      {r}
   700→                    </SelectItem>
   701→                  ))}
   702→                </SelectContent>
   703→              </Select>
   704→            </div>
   705→            <div className="grid gap-1.5">
   706→              <Label>Plan</Label>
   707→              <Select value={plan} onValueChange={setPlan}>
   708→                <SelectTrigger className="w-full">
   709→                  <SelectValue />
   710→                </SelectTrigger>
   711→                <SelectContent>
   712→                  {PLANS.map((p) => (
   713→                    <SelectItem key={p} value={p} className="capitalize">
   714→                      {p}
   715→                    </SelectItem>
   716→                  ))}
   717→                </SelectContent>
   718→              </Select>
   719→            </div>
   720→          </div>
   721→        </div>
   722→        <DialogFooter>
   723→          <DialogClose asChild>
   724→            <Button variant="outline" disabled={saving}>
   725→              Cancel
   726→            </Button>
   727→          </DialogClose>
   728→          <Button onClick={submit} disabled={saving} className={EMERALD_BTN}>
   729→            {saving && <Loader2 className="w-3.5 h-3.5 mr-1 animate-spin" />}
   730→            Create user
   731→          </Button>
   732→        </DialogFooter>
   733→      </DialogContent>
   734→    </Dialog>
   735→  );
   736→}
