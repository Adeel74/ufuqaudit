   111→   111→function fmtVolume(v: number): string {
   112→   112→  if (v >= 1_000_000) return (v / 1_000_000).toFixed(1) + "M";
   113→   113→  if (v >= 1_000) return (v / 1_000).toFixed(1) + "K";
   114→   114→  return v.toString();
   115→   115→}
   116→   116→
   117→   117→function fmtDelta(cur: number, prev: number): { text: string; delta: number } {
   118→   118→  const d = cur - prev;
   119→   119→  return { text: d === 0 ? "—" : (d > 0 ? `+${d}` : `${d}`), delta: d };
   120→   120→}
   121→   121→
   122→   122→function truncateUrl(u: string, n = 32): string {
   123→   123→  if (u.length <= n) return u;
   124→   124→  const stripped = u.replace(/^https?:\/\//, "");
   125→   125→  if (stripped.length <= n) return stripped;
   126→   126→  return stripped.slice(0, n - 1) + "…";
   127→   127→}

   208→   208→      const j = (await r.json()) as { keywords: KeywordRow[]; stats: KeywordStats };

   211→   211→      const msg = e instanceof Error ? e.message : "Failed to load keywords";
   212→   212→      setError(msg);
   213→   213→      setData({ keywords: [], stats: { total: 0, avgPosition: 0, totalVolume: 0, top3: 0, top10: 0, improved: 0 } });

   215→   215→      setLoading(false);
   216→   216→    }

   235→   235→      toast.error("Enter a keyword to track");
   236→   236→      return;
   237→   237→    }
   238→   238→    setAdding(true);
   239→   239→    try {
   240→   240→      const r = await fetch("/api/keywords", {
   241→   241→        method: "POST",
   242→   242→        headers: { "Content-Type": "application/json" },
   243→   243→        body: JSON.stringify({ keyword: kw }),
   244→   244→      });
   245→   245→      if (!r.ok) {
   246→   246→        const j = await r.json().catch(() => ({}));
   247→   247→        throw new Error(j?.error || `HTTP ${r.status}`);
   248→   248→      }
   249→   249→      const j = (await r.json()) as { keyword: KeywordRow };
   250→   250→      setData((prev) => {
   251→   251→        if (!prev) return prev;

   260→   260→              ([...others, j.keyword].reduce((s, k) => s + k.position, 0)) / (others.length + 1),

   264→   264→            totalVolume: [...others, j.keyword].reduce((s, k) => s + k.volume, 0),
   265→   265→            improved: [...others, j.keyword].filter((k) => k.position < k.prevPosition).length,
   266→   266→          },
   267→   267→        };
   268→   268→      });
   269→   269→      toast.success(`Now tracking "${kw}"`);
   270→   270→      setNewKw("");
   271→   271→      setAddOpen(false);
   272→   272→    } catch (e: unknown) {
   273→   273→      const msg = e instanceof Error ? e.message : "Failed to add keyword";
   274→   274→      toast.error(msg);
   275→   275→    } finally {
   276→   276→      setAdding(false);
   277→   277→    }
   278→   278→  };

   280→   280→  // ---------- Loading ----------

   306→   306→  }

   308→   308→  const keywords = data?.keywords ?? [];
