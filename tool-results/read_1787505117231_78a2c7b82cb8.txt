   126→   126→function toLocalInput(iso: string): string {
   127→   127→  if (!iso) return "";
   128→   128→  const d = new Date(iso);
   129→   129→  if (isNaN(d.getTime())) return "";
   130→   130→  const tz = d.getTimezoneOffset() * 60000;
   131→   131→  return new Date(d.getTime() - tz).toISOString().slice(0, 16);
   132→   132→}
