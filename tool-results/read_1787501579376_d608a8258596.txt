   119→   119→  const [paused, setPaused] = React.useState(false);
   120→   120→
   121→   121→  const load = React.useCallback(async () => {

   124→   124→      const r = await fetch("/api/admin/crawler").then(

   135→   135→
   136→   136→  React.useEffect(() => {
   137→   137→    void load();
   138→   138→  }, [load, refreshKey]);
   139→   139→
   140→   140→  const stats = data?.stats;
   141→   141→  const metrics = data?.metrics;
   142→   142→  const running = data?.running ?? [];
   143→   143→  const queued = data?.queued ?? [];
   144→   144→  const completed = data?.completed ?? [];
   145→   145→  const failed = data?.failed ?? [];
   146→   146→
   147→   147→  async function cancelCrawlJob() {

   150→   150→    await new Promise((r) => setTimeout(r, 500));
   151→   151→    setData((prev) =>
   152→   152→      prev
   153→   153→        ? { ...prev, running: prev.running.filter((c) => c.id !== cancelCrawl.id) }
   154→   154→        : prev,
   155→   155→    );
   156→   156→    toast.success(`Cancelled crawl ${truncate(cancelCrawl.id, 12)}`);
   157→   157→    setCancelCrawl(null);
   158→   158→    setCancelling(false);
   159→   159→  }
   160→   160→
   161→   161→  function cancelQueued(item: QueuedCrawl) {
   162→   162→    setData((prev) =>
   163→   163→      prev
   164→   164→        ? { ...prev, queued: prev.queued.filter((c) => c.id !== item.id) }
   165→   165→        : prev,
   166→   166→    );
   167→   167→    toast.success(`Removed queued crawl ${truncate(item.id, 12)}`);
   168→   168→  }
   169→   169→
   170→   170→  function retryFailed(item: FailedCrawl) {
   171→   171→    toast.success(`Re-enqueued failed crawl ${truncate(item.id, 12)}`);
   172→   172→    setData((prev) =>
   173→   173→      prev
   174→   174→        ? {
   175→   175→            ...prev,
   176→   176→            failed: prev.failed.filter((c) => c.id !== item.id),
   177→   177→            queued: [
   178→   178→              ...prev.queued,
   179→   179→              {
   180→   180→                id: item.id,
   181→   181→                url: item.url,
   182→   182→                userId: item.userId,
   183→   183→                position: prev.queued.length + 1,
   184→   184→                enqueuedAt: new Date().toISOString(),
   185→   185→              },
   186→   186→            ],
   187→   187→          }
   188→   188→        : prev,
   189→   189→    );
   190→   190→  }
   191→   191→
   192→   192→  return (
   193→   193→    <div className="space-y-6">

   216→   216→          <Cpu className="w-4 h-4 text-emerald-600" /> Server Metrics

   218→   218→        {loading || !metrics ? (

   225→   225→              value={metrics.cpu}
   226→   226→              max={100}
   227→   227→              suffix="%"
   228→   228→              current={`${metrics.cpu}%`}
   229→   229→            />
   230→   230→            <MetricBar
   231→   231→              icon={MemoryStick}
   232→   232→              label="Memory"
   233→   233→              value={metrics.memory}
   234→   234→              max={metrics.memoryLimit}
   235→   235→              suffix="MB"
   236→   236→              current={`${metrics.memory} / ${formatCompact(metrics.memoryLimit)} MB`}

   253→   253→            />

   261→   261→            />
   262→   262→            <MetricBar
   263→   263→              icon={Database}

   265→   265→              value={metrics.dbConnections}

   269→   269→            />

   271→   271→              <div className="flex items-center justify-between gap-2 p-3 rounded-lg border bg-card">
   272→   272→                <div className="flex items-center gap-2">
   273→   273→                  <Network className="w-4 h-4 text-emerald-600" />

   275→   275→                </div>
   276→   276→                <Badge variant="outline" className="tabular-nums bg-emerald-50 text-emerald-700 border-emerald-200 dark:bg-emerald-950/40 dark:text-emerald-300 dark:border-emerald-800">
   277→   277→                  {metrics.network.toFixed(1)} Mbps
   278→   278→                </Badge>
   279→   279→              </div>
   280→   280→            </div>
   281→   281→          </div>
   282→   282→        )}

   284→   284→
   285→   285→      {/* Controls */}

   288→   288→          <Server className="w-4 h-4 text-emerald-600" /> Crawler Controls
   289→   289→        </h3>

   298→   298→          </Button>
   299→   299→          <Button

   303→   303→          >

   305→   305→          </Button>
   306→   306→          <Button

   309→   309→            onClick={() => toast.info("Stop signal sent (demo)")}
   310→   310→          >

   313→   313→        </div>

   318→   318→              <Label className="text-sm">Max concurrent crawlers</Label>
   319→   319→              <Badge variant="outline" className="tabular-nums">{maxConcurrent}</Badge>
   320→   320→            </div>
   321→   321→            <Slider
   322→   322→              value={[maxConcurrent]}
   323→   323→              min={1}
   324→   324→              max={20}
   325→   325→              step={1}
   326→   326→              onValueChange={(arr) => setMaxConcurrent(arr[0] ?? maxConcurrent)}
   327→   327→            />
   328→   328→            <div className="flex items-center justify-between text-[10px] text-muted-foreground">
   329→   329→              <span>1</span>
   330→   330→              <span>20</span>
   331→   331→            </div>
   332→   332→          </div>
   333→   333→          <div className="grid gap-2">
   334→   334→            <div className="flex items-center justify-between">
   335→   335→              <Label className="text-sm">Crawl timeout</Label>
   336→   336→              <Badge variant="outline" className="tabular-nums">{crawlTimeout.toLocaleString()} ms</Badge>
   337→   337→            </div>
   338→   338→            <Slider
   339→   339→              value={[crawlTimeout]}
   340→   340→              min={1000}
   341→   341→              max={60000}

   343→   343→              onValueChange={(arr) => setCrawlTimeout(arr[0] ?? crawlTimeout)}
   344→   344→            />
   345→   345→            <div className="flex items-center justify-between text-[10px] text-muted-foreground">
   346→   346→              <span>1,000 ms</span>
   347→   347→              <span>60,000 ms</span>
   348→   348→            </div>
   349→   349→          </div>
   350→   350→        </div>
   351→   351→      </Card>
   352→   352→
   353→   353→      {/* Running crawlers table */}
   354→   354→      <Card className="p-5">

   357→   357→          <Badge variant="secondary" className="ml-1">{running.length}</Badge>

   359→   359→            <Badge variant="outline" className="bg-amber-50 text-amber-700 border-amber-200 dark:bg-amber-950/40 dark:text-amber-300 dark:border-amber-800">
   360→   360→              Paused
   361→   361→            </Badge>

   363→   363→        </h3>
   364→   364→        <div className={`-mx-2 px-2 ${SCROLLBAR_CLS} max-h-[50vh]`}>

   661→   661→function MetricBar({
   662→   662→  icon: Icon,
   663→   663→  label,
   664→   664→  value,
   665→   665→  max,
   666→   666→  suffix,
   667→   667→  current,
   668→   668→}: {
   669→   669→  icon: React.ComponentType<{ className?: string }>;
   670→   670→  label: string;
   671→   671→  value: number;
   672→   672→  max: number;
   673→   673→  suffix: string;
   674→   674→  current: string;
   675→   675→}) {
   676→   676→  const pct = max > 0 ? Math.min(100, (value / max) * 100) : 0;
   677→   677→  const barColor =
   678→   678→    pct >= 90
   679→   679→      ? "bg-red-500"
   680→   680→      : pct >= 70
   681→   681→        ? "bg-amber-500"
   682→   682→        : "bg-emerald-500";
   683→   683→  const textColor =
   684→   684→    pct >= 90
   685→   685→      ? "text-red-600 dark:text-red-400"
   686→   686→      : pct >= 70
   687→   687→        ? "text-amber-600 dark:text-amber-400"
   688→   688→        : "text-emerald-600 dark:text-emerald-400";
   689→   689→  return (
   690→   690→    <div className="p-3 rounded-lg border bg-card">
   691→   691→      <div className="flex items-center justify-between mb-2">
   692→   692→        <div className="flex items-center gap-2">
   693→   693→          <Icon className="w-4 h-4 text-muted-foreground" />
   694→   694→          <span className="text-sm font-medium">{label}</span>
   695→   695→        </div>
   696→   696→        <span className={`text-xs tabular-nums font-mono ${textColor}`}>
   697→   697→          {current}
   698→   698→        </span>
   699→   699→      </div>
   700→   700→      <div className="relative w-full h-2 rounded-full bg-muted overflow-hidden">
   701→   701→        <div
   702→   702→          className={`absolute top-0 left-0 h-full transition-all ${barColor}`}
   703→   703→          style={{ width: `${pct}%` }}
   704→   704→        />
   705→   705→      </div>
   706→   706→      {suffix && (
   707→   707→        <div className="text-[10px] text-muted-foreground mt-1">
   708→   708→          {pct.toFixed(0)}% of {max.toLocaleString()} {suffix}
   709→   709→        </div>
   710→   710→      )}
   711→   711→    </div>
   712→   712→  );
   713→   713→}
