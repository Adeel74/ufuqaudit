    90→// ---------- Helpers ----------
    91→function posColor(p: number): string {
    92→  if (p <= 3) return "#10b981";
    93→  if (p <= 10) return "#f59e0b";
    94→  if (p <= 30) return "#f97316";

    98→function posBadgeCls(p: number): string {
    99→  if (p <= 3) return "bg-emerald-100 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-400";
   100→  if (p <= 10) return "bg-amber-100 text-amber-700 dark:bg-amber-950/40 dark:text-amber-400";
   101→  if (p <= 30) return "bg-orange-100 text-orange-700 dark:bg-orange-950/40 dark:text-orange-400";
   102→  return "bg-red-100 text-red-700 dark:bg-red-950/40 dark:text-red-400";
   103→}
   104→
   105→function diffColor(d: number): string {
   106→  if (d < 0) return "text-emerald-600 dark:text-emerald-400"; // improved (position dropped)
   107→  if (d > 0) return "text-red-600 dark:text-red-400"; // dropped
   108→  return "text-muted-foreground";
   109→}
   110→
   111→function fmtVolume(v: number): string {
   112→  if (v >= 1_000_000) return (v / 1_000_000).toFixed(1) + "M";
   113→  if (v >= 1_000) return (v / 1_000).toFixed(1) + "K";
   114→  return v.toString();
   115→}
   116→
   117→function fmtDelta(cur: number, prev: number): { text: string; delta: number } {
   118→  const d = cur - prev;
   119→  return { text: d === 0 ? "—" : (d > 0 ? `+${d}` : `${d}`), delta: d };
   120→}
   121→
   122→function truncateUrl(u: string, n = 32): string {
   123→  if (u.length <= n) return u;
   124→  const stripped = u.replace(/^https?:\/\//, "");
   125→  if (stripped.length <= n) return stripped;
   126→  return stripped.slice(0, n - 1) + "…";
   127→}

   152→  label, sortKey, current, dir, onSort, className, align = "left",
   153→}: {
   154→  label: string;
   155→  sortKey: SortKey;
   156→  current: SortKey;

   158→  onSort: (k: SortKey) => void;
   159→  className?: string;

   163→  return (
   164→    <TableHead className={className}>
   165→      <button
   166→        type="button"
   167→        onClick={() => onSort(sortKey)}
   168→        className={cn(
   169→          "inline-flex items-center gap-1 text-xs font-medium hover:text-foreground transition-colors",
   170→          align === "right" && "flex-row-reverse",
   171→          active && "text-foreground",

   185→// =================== Component ===================
   186→export function KeywordsView() {
   187→  const [data, setData] = React.useState<{ keywords: KeywordRow[]; stats: KeywordStats } | null>(null);
   188→  const [error, setError] = React.useState<string | null>(null);
   189→  const [loading, setLoading] = React.useState(true);
   190→
   191→  // filter + sort state
   192→  const [search, setSearch] = React.useState("");
   193→  const [range, setRange] = React.useState<RangeFilter>("all");
   194→  const [sortKey, setSortKey] = React.useState<SortKey>("position");
   195→  const [sortDir, setSortDir] = React.useState<SortDir>("asc");
   196→
   197→  // add-keyword dialog
   198→  const [addOpen, setAddOpen] = React.useState(false);
   199→  const [newKw, setNewKw] = React.useState("");
   200→  const [adding, setAdding] = React.useState(false);
   201→
   202→  const fetchData = React.useCallback(async () => {
   203→    setLoading(true);
   204→    setError(null);
   205→    try {
   206→      const r = await fetch("/api/keywords", { cache: "no-store" });
   207→      if (!r.ok) throw new Error(`HTTP ${r.status}`);
   208→      const j = (await r.json()) as { keywords: KeywordRow[]; stats: KeywordStats };
   209→      setData({ keywords: Array.isArray(j.keywords) ? j.keywords : [], stats: j.stats });
   210→    } catch (e: unknown) {
   211→      const msg = e instanceof Error ? e.message : "Failed to load keywords";
   212→      setError(msg);
   213→      setData({ keywords: [], stats: { total: 0, avgPosition: 0, totalVolume: 0, top3: 0, top10: 0, improved: 0 } });
   214→    } finally {
   215→      setLoading(false);
   216→    }
   217→  }, []);
   218→
   219→  React.useEffect(() => {
   220→    fetchData();
   221→  }, [fetchData]);
   222→
   223→  const onSort = (k: SortKey) => {
   224→    if (k === sortKey) {
   225→      setSortDir((d) => (d === "asc" ? "desc" : "asc"));
   226→    } else {
   227→      setSortKey(k);
   228→      setSortDir(k === "keyword" ? "asc" : "asc");
   229→    }
   230→  };
   231→
   232→  const onAddKeyword = async () => {
   233→    const kw = newKw.trim();
   234→    if (!kw) {
   235→      toast.error("Enter a keyword to track");
   236→      return;
   237→    }
   238→    setAdding(true);
   239→    try {
   240→      const r = await fetch("/api/keywords", {
   241→        method: "POST",
   242→        headers: { "Content-Type": "application/json" },
   243→        body: JSON.stringify({ keyword: kw }),
   244→      });
   245→      if (!r.ok) {
   246→        const j = await r.json().catch(() => ({}));
   247→        throw new Error(j?.error || `HTTP ${r.status}`);
   248→      }
   249→      const j = (await r.json()) as { keyword: KeywordRow };
   250→      setData((prev) => {
   251→        if (!prev) return prev;
   252→        // Avoid duplicates — replace if same keyword exists.
   253→        const others = prev.keywords.filter((k) => k.keyword.toLowerCase() !== kw.toLowerCase());
   254→        return {
   255→          keywords: [j.keyword, ...others],
   256→          stats: {
   257→            ...prev.stats,
   258→            total: prev.stats.total + 1,
   259→            avgPosition: Math.round(
   260→              ([...others, j.keyword].reduce((s, k) => s + k.position, 0)) / (others.length + 1),
   261→            ),
   262→            top3: [...others, j.keyword].filter((k) => k.position <= 3).length,
   263→            top10: [...others, j.keyword].filter((k) => k.position <= 10).length,
   264→            totalVolume: [...others, j.keyword].reduce((s, k) => s + k.volume, 0),
   265→            improved: [...others, j.keyword].filter((k) => k.position < k.prevPosition).length,
   266→          },
   267→        };
   268→      });
   269→      toast.success(`Now tracking "${kw}"`);
   270→      setNewKw("");
   271→      setAddOpen(false);
   272→    } catch (e: unknown) {
   273→      const msg = e instanceof Error ? e.message : "Failed to add keyword";
   274→      toast.error(msg);
   275→    } finally {
   276→      setAdding(false);
   277→    }
   278→  };
   279→
   280→  // ---------- Loading ----------
   281→  if (loading) {
   282→    return (
   283→      <div className="space-y-6">
   284→        <ViewHeader
   285→          title="Keyword Rank Tracker"
   286→          subtitle="Monitor your keyword positions and SERP features"
   287→          icon={Search}
   288→          actions={
   289→            <Button size="sm" className={EMERALD_BTN} disabled>
   290→              <Plus className="w-3.5 h-3.5 mr-1" /> Add keyword
   291→            </Button>
   292→          }
   293→        />
   294→        <div className="grid grid-cols-2 lg:grid-cols-5 gap-4">
   295→          {[0, 1, 2, 3, 4].map((i) => (
   296→            <Skeleton key={i} className="h-24 rounded-xl" />
   297→          ))}
   298→        </div>
   299→        <div className="grid grid-cols-1 lg:grid-cols-2 gap-4">
   300→          <Skeleton className="h-[300px] rounded-xl" />
   301→          <Skeleton className="h-[300px] rounded-xl" />
   302→        </div>
   303→        <Skeleton className="h-96 rounded-xl" />
   304→      </div>
   305→    );
   306→  }
   307→
   308→  const keywords = data?.keywords ?? [];
   309→  const stats = data?.stats ?? { total: 0, avgPosition: 0, totalVolume: 0, top3: 0, top10: 0, improved: 0 };
   310→
   311→  // ---------- Empty ----------
   312→  if (keywords.length === 0) {
   313→    return (
   314→      <div className="space-y-6">
   315→        <ViewHeader
   316→          title="Keyword Rank Tracker"
   317→          subtitle="Monitor your keyword positions and SERP features"
   318→          icon={Search}
   319→          actions={
   320→            <Button size="sm" className={EMERALD_BTN} onClick={() => setAddOpen(true)}>
   321→              <Plus className="w-3.5 h-3.5 mr-1" /> Add keyword
   322→            </Button>
   323→          }
   324→        />
   325→        <Card className="p-10 text-center border-dashed">
   326→          <div className="w-14 h-14 mx-auto rounded-full bg-muted flex items-center justify-center mb-3">
   327→            <Search className="w-6 h-6 text-muted-foreground" />
   328→          </div>
   329→          <h3 className="font-semibold text-lg">No keywords tracked yet</h3>
   330→          <p className="text-sm text-muted-foreground mt-1 max-w-md mx-auto">
   331→            {error
   332→              ? `We couldn't load your keywords (${error}). Try again in a moment.`
   333→              : "Add your first keyword to start tracking its position, search volume, and SERP features."}
   334→          </p>
   335→          <Button className={cn(EMERALD_BTN, "mt-5")} onClick={() => setAddOpen(true)}>
   336→            <Plus className="w-4 h-4 mr-1" /> Add your first keyword
   337→          </Button>
   338→        </Card>
   339→        <AddKeywordDialog
   340→          open={addOpen}
   341→          onOpenChange={setAddOpen}
   342→          value={newKw}
   343→          onChange={setNewKw}
   344→          onAdd={onAddKeyword}
   345→          adding={adding}
   346→        />
   347→      </div>
   348→    );
   349→  }
   350→
   351→  // ---------- Derived: distribution ----------
   352→  const distribution = BUCKETS.map((b) => ({
   353→    name: b.label,
   354→    key: b.key,
   355→    value: keywords.filter((k) => b.test(k.position)).length,
   356→    color: b.color,
   357→  }));
   358→
   359→  // ---------- Derived: top-5 keyword trends (12 weeks) ----------
   360→  // Lower position = better, so we pick the 5 keywords with the lowest current position.
   361→  const top5 = [...keywords].sort((a, b) => a.position - b.position).slice(0, 5);
   362→  // Each keyword's trend array has 13 points (12 weeks + current). Use index as x-axis.
   363→  const weeks = top5[0]?.trend.length ?? 0;
   364→  const trendData = Array.from({ length: weeks }, (_, i) => {
   365→    const row: Record<string, number | string> = { week: `W${i + 1}` };
   366→    top5.forEach((k) => {
   367→      row[k.keyword] = k.trend[i] ?? 0;
   368→    });
   369→    return row;
   370→  });
   371→
   372→  // ---------- Derived: filtered + sorted table rows ----------
   373→  const filtered = keywords.filter((k) => {
   374→    if (search && !k.keyword.toLowerCase().includes(search.toLowerCase())) return false;
   375→    if (range === "top3" && k.position > 3) return false;
   376→    if (range === "top10" && k.position > 10) return false;
   377→    if (range === "top30" && k.position > 30) return false;
   378→    return true;
   379→  });
   380→
   381→  const sorted = [...filtered].sort((a, b) => {
   382→    let cmp = 0;
   383→    if (sortKey === "keyword") cmp = a.keyword.localeCompare(b.keyword);
   384→    else cmp = (a[sortKey] as number) - (b[sortKey] as number);
   385→    return sortDir === "asc" ? cmp : -cmp;
   386→  });
   387→
   388→  return (
   389→    <div className="space-y-6">
   390→      <ViewHeader
   391→        title="Keyword Rank Tracker"
   392→        subtitle="Monitor your keyword positions and SERP features"
   393→        icon={Search}
   394→        actions={
   395→          <Button size="sm" className={EMERALD_BTN} onClick={() => setAddOpen(true)}>
   396→            <Plus className="w-3.5 h-3.5 mr-1" /> Add keyword
   397→          </Button>
   398→        }
   399→      />
   400→
   401→      {/* Stats row — 5 KPI cards */}
   402→      <div className="grid grid-cols-2 lg:grid-cols-5 gap-4">
   403→        <StatCard label="Total Keywords" value={stats.total} hint="Tracked" color="#0f766e" icon={Hash} />
   404→        <StatCard label="Avg Position" value={stats.avgPosition} hint="Across all keywords" color={posColor(stats.avgPosition)} icon={Gauge} />
   405→        <StatCard label="Top 3" value={stats.top3} hint="On page 1, top spots" color="#10b981" icon={Trophy} />
   406→        <StatCard label="Top 10" value={stats.top10} hint="On page 1" color="#f59e0b" icon={Target} />
   407→        <StatCard
   408→          label="Improved"
   409→          value={stats.improved}
   410→          hint="Vs. previous check"
   411→          color="#10b981"
   412→          icon={TrendingUp}
   413→        />
   414→      </div>
   415→
   416→      {/* Distribution donut + Trend lines */}
   417→      <div className="grid grid-cols-1 lg:grid-cols-2 gap-4">
   418→        {/* Distribution donut */}
   419→        <Card className="p-5">
   420→          <div className="flex items-center justify-between mb-4">
   421→            <div>
   422→              <h3 className="font-semibold flex items-center gap-2">
   423→                <span className="w-2 h-2 rounded-full" style={{ backgroundColor: EMERALD }} />
   424→                Ranking Distribution
   425→              </h3>
   426→              <p className="text-xs text-muted-foreground mt-0.5">Keywords by SERP position bucket</p>
   427→            </div>
   428→            <Badge variant="outline" className="font-mono text-[11px]">{keywords.length}</Badge>
   429→          </div>
   430→          <div style={{ height: 240 }}>
   431→            <ResponsiveContainer width="100%" height="100%">
   432→              <PieChart>
   433→                <Pie
   434→                  data={distribution}
   435→                  dataKey="value"
   436→                  nameKey="name"
   437→                  cx="50%"
   438→                  cy="50%"
   439→                  innerRadius={55}
   440→                  outerRadius={85}
   441→                  paddingAngle={2}
   442→                  stroke="none"
   443→                >
   444→                  {distribution.map((d) => (
   445→                    <Cell key={d.key} fill={d.color} />
   446→                  ))}
   447→                </Pie>
   448→                <RTooltip
   449→                  contentStyle={{
   450→                    borderRadius: 10, border: "1px solid #e2e8f0", fontSize: 12, padding: "8px 10px",
   451→                  }}
   452→                  formatter={(v: number, n: string) => [`${v} keyword${v === 1 ? "" : "s"}`, n]}
   453→                />
   454→              </PieChart>
   455→            </ResponsiveContainer>
   456→          </div>
   457→          {/* Legend */}
   458→          <div className="grid grid-cols-2 sm:grid-cols-3 gap-2 mt-3">
   459→            {distribution.map((d) => (
   460→              <div key={d.key} className="flex items-center gap-2 text-xs">
   461→                <span className="w-2.5 h-2.5 rounded-sm shrink-0" style={{ backgroundColor: d.color }} />
   462→                <span className="text-muted-foreground truncate">{d.name}</span>
   463→                <span className="ml-auto font-semibold tabular-nums">{d.value}</span>
   464→              </div>
   465→            ))}
   466→          </div>
   467→        </Card>
   468→
   469→        {/* Top-5 position trend (lower = better) */}
   470→        <Card className="p-5">
   471→          <div className="flex items-center justify-between mb-4">
   472→            <div>
   473→              <h3 className="font-semibold">Top 5 Keyword Trends</h3>
   474→              <p className="text-xs text-muted-foreground mt-0.5">
   475→                Position over 12 weeks — <span className="text-emerald-600 dark:text-emerald-400 font-medium">lower is better</span>
   476→              </p>
   477→            </div>
   478→            <Badge variant="outline" className="font-mono text-[11px]">12 wk</Badge>
   479→          </div>
   480→          <div style={{ height: 280 }}>
   481→            <ResponsiveContainer width="100%" height="100%">
   482→              <LineChart data={trendData} margin={{ top: 8, right: 16, left: -8, bottom: 0 }}>

   486→                <YAxis

   656→              })}
   657→              {sorted.length === 0 && (

   664→            </TableBody>

   666→        </div>

   669→      <AddKeywordDialog

   674→        onAdd={onAddKeyword}

   677→    </div>

   681→// ---------- Difficulty bar (0–100, colored) ----------
   682→function DifficultyBar({ value }: { value: number }) {
   683→  const v = Math.max(0, Math.min(100, value));
   684→  const color = v <= 30 ? "#10b981" : v <= 60 ? "#f59e0b" : v <= 80 ? "#f97316" : "#ef4444";
   685→  const label = v <= 30 ? "Easy" : v <= 60 ? "Medium" : v <= 80 ? "Hard" : "Very hard";
   686→  return (
   687→    <div className="flex items-center gap-2 justify-end">
   688→      <div className="flex-1 max-w-[80px] h-1.5 rounded-full bg-muted overflow-hidden">
   689→        <div
   690→          className="h-full rounded-full transition-all"
   691→          style={{ width: `${v}%`, backgroundColor: color }}
   692→        />
   693→      </div>
   694→      <span className="text-xs tabular-nums w-7 text-right" style={{ color }}>{v}</span>
   695→      <span className="text-[10px] text-muted-foreground w-14 hidden sm:inline">{label}</span>
   696→    </div>
   697→  );
   698→}
