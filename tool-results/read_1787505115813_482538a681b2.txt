   126→function toLocalInput(iso: string): string {
   127→  if (!iso) return "";
   128→  const d = new Date(iso);
   129→  if (isNaN(d.getTime())) return "";
   130→  const tz = d.getTimezoneOffset() * 60000;
   131→  return new Date(d.getTime() - tz).toISOString().slice(0, 16);
   132→}

   408→          </AlertDialogHeader>
   409→          <AlertDialogFooter>
   410→            <AlertDialogCancel disabled={deleting}>Cancel</AlertDialogCancel>

   417→              className="bg-red-600 hover:bg-red-700 text-white"

   422→          </AlertDialogFooter>
   423→        </AlertDialogContent>

   425→
   426→      <AlertDialog open={!!endNow} onOpenChange={(o) => { if (!o) setEndNow(null); }}>
   427→        <AlertDialogContent>
   428→          <AlertDialogHeader>

   430→            <AlertDialogDescription>
   431→              <span className="font-medium text-foreground">{endNow?.title}</span> will be immediately marked as ended.
   432→              Users will no longer see the banner.
   433→            </AlertDialogDescription>
   434→          </AlertDialogHeader>
   435→          <AlertDialogFooter>
   436→            <AlertDialogCancel disabled={ending}>Keep active</AlertDialogCancel>

   443→              className="bg-amber-600 hover:bg-amber-700 text-white"

   446→              End now
   447→            </AlertDialogAction>
   448→          </AlertDialogFooter>
   449→        </AlertDialogContent>
   450→      </AlertDialog>
   451→    </div>
   452→  );
   453→}

   512→  open: boolean;
   513→  onOpenChange: (o: boolean) => void;
   514→  onCreated: (a: Announcement) => void;
   515→  existing: Announcement | null;
   516→}) {
   517→  const [title, setTitle] = React.useState("");
   518→  const [description, setDescription] = React.useState("");
   519→  const [type, setType] = React.useState<AnnouncementType>("info");
   520→  const [audience, setAudience] = React.useState<Audience>("all");
   521→  const [chDashboard, setChDashboard] = React.useState(true);
   522→  const [chEmail, setChEmail] = React.useState(false);
   523→  const [chPopup, setChPopup] = React.useState(false);
   524→  const [startDate, setStartDate] = React.useState("");
   525→  const [endDate, setEndDate] = React.useState("");
   526→  const [ctaLabel, setCtaLabel] = React.useState("");
   527→  const [ctaHref, setCtaHref] = React.useState("");
   528→  const [saving, setSaving] = React.useState(false);
   529→
   530→  React.useEffect(() => {
   531→    if (!open) return;
   532→    if (existing) {
   533→      setTitle(existing.title);
   534→      setDescription(existing.description);
   535→      setType(existing.type);
   536→      setAudience(existing.audience);
   537→      setChDashboard(existing.channels.dashboard);
   538→      setChEmail(existing.channels.email);
   539→      setChPopup(existing.channels.popup);
   540→      setStartDate(toLocalInput(existing.startDate));
   541→      setEndDate(toLocalInput(existing.endDate));
   542→      setCtaLabel(existing.ctaLabel);
   543→      setCtaHref(existing.ctaHref);
   544→    } else {
   545→      setTitle("");
   546→      setDescription("");
   547→      setType("info");
   548→      setAudience("all");
   549→      setChDashboard(true);
   550→      setChEmail(false);
   551→      setChPopup(false);
   552→      const now = new Date();
   553→      const later = new Date(now.getTime() + 7 * 86400000);
   554→      setStartDate(toLocalInput(now.toISOString()));
   555→      setEndDate(toLocalInput(later.toISOString()));
   556→      setCtaLabel("Learn more");
   557→      setCtaHref("#");
   558→    }
   559→  }, [open, existing]);
   560→
   561→  async function submit() {
   562→    if (!title.trim()) {
   563→      toast.error("Title required");
   564→      return;
   565→    }
   566→    setSaving(true);
   567→    try {
   568→      const body = {
   569→        title: title.trim(),
   570→        description: description.trim(),
   571→        type,
   572→        audience,
   573→        channels: { dashboard: chDashboard, email: chEmail, popup: chPopup },
   574→        startDate: startDate ? new Date(startDate).toISOString() : new Date().toISOString(),
   575→        endDate: endDate ? new Date(endDate).toISOString() : new Date(Date.now() + 7 * 86400000).toISOString(),
   576→        ctaLabel: ctaLabel.trim(),
   577→        ctaHref: ctaHref.trim() || "#",
   578→      };
   579→      const res = await fetch("/api/admin/announcements", {
   580→        method: "POST",
   581→        headers: { "Content-Type": "application/json" },
   582→        body: JSON.stringify(body),
   583→      });
   584→      let created: Announcement | null = null;
   585→      if (res.ok) {
   586→        const json = (await res.json()) as { announcement: Announcement };
   587→        created = json.announcement;
   588→      } else {
   589→        created = {
   590→          id: existing?.id ?? `ann_${Date.now()}`,
   591→          ...body,
   592→          status: "scheduled",
   593→          createdAt: new Date().toISOString(),
   594→          views: existing?.views ?? 0,
   595→          clicks: existing?.clicks ?? 0,
   596→        };
   597→      }
   598→      onCreated(created);
   599→    } catch (e) {
   600→      console.error(e);
   601→      toast.error("Failed to save announcement");
   602→    } finally {
   603→      setSaving(false);
   604→    }
   605→  }
   606→
   607→  const TIcon = TYPE_ICON[type];
   608→  const tColor = TYPE_COLOR_HEX[type];
   609→
   610→  return (
   611→    <Dialog open={open} onOpenChange={onOpenChange}>
   612→      <DialogContent className="sm:max-w-2xl max-h-[90vh] overflow-y-auto">
   613→        <DialogHeader>
   614→          <DialogTitle>{existing ? `Edit announcement` : "Create announcement"}</DialogTitle>
   615→        </DialogHeader>
   616→        <div className="grid gap-3">
   617→          <div className="grid gap-1.5">
   618→            <Label htmlFor="an-title">Title</Label>
   619→            <Input
   620→              id="an-title"
   621→              placeholder="AEO scoring is now live!"
   622→              value={title}
   623→              onChange={(e) => setTitle(e.target.value)}
   624→            />
   625→          </div>
   626→          <div className="grid gap-1.5">
   627→            <Label htmlFor="an-desc">Description</Label>
   628→            <Textarea
   629→              id="an-desc"
   630→              placeholder="Short message users will see in the banner."
   631→              value={description}
   632→              onChange={(e) => setDescription(e.target.value)}
   633→              rows={3}
   634→            />
   635→          </div>
   636→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   637→            <div className="grid gap-1.5">
   638→              <Label>Type</Label>
   639→              <Select value={type} onValueChange={(v) => setType(v as AnnouncementType)}>
   640→                <SelectTrigger className="w-full">
   641→                  <SelectValue />
   642→                </SelectTrigger>
   643→                <SelectContent>
   644→                  <SelectItem value="info">Info</SelectItem>
   645→                  <SelectItem value="success">Success</SelectItem>
   646→                  <SelectItem value="warning">Warning</SelectItem>
   647→                  <SelectItem value="maintenance">Maintenance</SelectItem>
   648→                </SelectContent>
   649→              </Select>
   650→            </div>
   651→            <div className="grid gap-1.5">
   652→              <Label>Audience</Label>
   653→              <Select value={audience} onValueChange={(v) => setAudience(v as Audience)}>
   654→                <SelectTrigger className="w-full">
   655→                  <SelectValue />
   656→                </SelectTrigger>
   657→                <SelectContent>
   658→                  <SelectItem value="all">All users</SelectItem>
   659→                  <SelectItem value="free">Free</SelectItem>
   660→                  <SelectItem value="starter">Starter</SelectItem>
   661→                  <SelectItem value="pro">Pro</SelectItem>
   662→                  <SelectItem value="agency">Agency</SelectItem>
   663→                  <SelectItem value="trial">Trial</SelectItem>
   664→                </SelectContent>
   665→              </Select>
   666→            </div>
   667→          </div>
   668→          <Separator />
   669→          <div>
   670→            <Label className="text-sm mb-2 block">Channels</Label>
   671→            <div className="grid grid-cols-3 gap-2">
   672→              <ChannelSwitch
   673→                label="Dashboard"
   674→                icon={Eye}
   675→                checked={chDashboard}
   676→                onCheckedChange={setChDashboard}
   677→              />
   678→              <ChannelSwitch
   679→                label="Email"
   680→                icon={Mail}
   681→                checked={chEmail}
   682→                onCheckedChange={setChEmail}
   683→              />
   684→              <ChannelSwitch
   685→                label="Popup"
   686→                icon={Bell}
   687→                checked={chPopup}
   688→                onCheckedChange={setChPopup}
   689→              />
   690→            </div>
   691→          </div>
   692→          <Separator />
   693→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   694→            <div className="grid gap-1.5">
   695→              <Label htmlFor="an-start">Start date</Label>
   696→              <Input
   697→                id="an-start"
   698→                type="datetime-local"
   699→                value={startDate}
   700→                onChange={(e) => setStartDate(e.target.value)}
   701→              />
   702→            </div>
   703→            <div className="grid gap-1.5">
   704→              <Label htmlFor="an-end">End date</Label>
   705→              <Input
   706→                id="an-end"
   707→                type="datetime-local"
   708→                value={endDate}
   709→                onChange={(e) => setEndDate(e.target.value)}
   710→              />
   711→            </div>
   712→          </div>
   713→          <Separator />
   714→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   715→            <div className="grid gap-1.5">
   716→              <Label htmlFor="an-cta-label">CTA label</Label>
   717→              <Input
   718→                id="an-cta-label"
   719→                placeholder="Learn more"
   720→                value={ctaLabel}
   721→                onChange={(e) => setCtaLabel(e.target.value)}
   722→              />
   723→            </div>
   724→            <div className="grid gap-1.5">
   725→              <Label htmlFor="an-cta-href">CTA href</Label>
   726→              <Input
   727→                id="an-cta-href"
   728→                placeholder="/dashboard"
   729→                value={ctaHref}
   730→                onChange={(e) => setCtaHref(e.target.value)}
   731→              />
   732→            </div>
   733→          </div>
   734→          <Separator />
   735→          <div className="rounded-lg border p-3 bg-muted/30">
   736→            <div className="text-[10px] text-muted-foreground mb-2">Live preview</div>
   737→            <div
   738→              className="rounded-md border p-3"
   739→              style={{ backgroundColor: `${tColor}10`, borderColor: `${tColor}40` }}
   740→            >
   741→              <div className="flex items-center gap-2">
   742→                <div
   743→                  className="w-7 h-7 rounded-md flex items-center justify-center shrink-0"
   744→                  style={{ backgroundColor: `${tColor}25`, color: tColor }}
   745→                >
   746→                  <TIcon className="w-3.5 h-3.5" />
   747→                </div>
   748→                <div className="font-semibold text-sm" style={{ color: tColor }}>
   749→                  {title || "Untitled announcement"}
   750→                </div>
   751→              </div>
   752→              <p className="text-xs text-foreground/80 mt-2 line-clamp-2">
   753→                {description || "Description preview appears here."}
   754→              </p>
   755→              {ctaLabel && (
   756→                <button
   757→                  className="mt-2 text-xs font-medium px-2 py-1 rounded-md text-white"
   758→                  style={{ backgroundColor: tColor }}
   759→                  type="button"
   760→                >
   761→                  {ctaLabel}
   762→                </button>
   763→              )}
   764→            </div>
   765→          </div>
   766→        </div>
   767→        <DialogFooter>
   768→          <DialogClose asChild>
   769→            <Button variant="outline" disabled={saving}>Cancel</Button>
   770→          </DialogClose>
   771→          <Button onClick={submit} disabled={saving} className={EMERALD_BTN}>
   772→            {saving && <Loader2 className="w-3.5 h-3.5 mr-1 animate-spin" />}
   773→            {existing ? "Save changes" : "Schedule announcement"}
   774→          </Button>
   775→        </DialogFooter>
   776→      </DialogContent>
   777→    </Dialog>
   778→  );
   779→}

   782→  label,

   784→  checked,
   785→  onCheckedChange,
   786→}: {
   787→  label: string;
   788→  icon: React.ComponentType<{ className?: string }>;
   789→  checked: boolean;
   790→  onCheckedChange: (v: boolean) => void;
   791→}) {

   794→      <Switch checked={checked} onCheckedChange={onCheckedChange} id={`ch-${label}`} />

   796→        <Icon className="w-3 h-3" /> {label}
   797→      </Label>
   798→    </div>
   799→  );
   800→}
