    81→
    82→interface EnrichedUser extends AdminUser {
    83→  status: "active" | "suspended";

    90→
    91→function enrichUsers(users: AdminUser[]): EnrichedUser[] {
    92→  // Mock enrichment — would come from richer API in production
    93→  return users.map((u, i) => {
    94→    const seed = u.id.charCodeAt(0) + u.id.charCodeAt(u.id.length - 1) + i;
    95→    return {
    96→      ...u,
    97→      status: u.role === "admin" ? "active" : seed % 7 === 0 ? "suspended" : "active",
    98→      projects: seed % 12,
    99→      audits: seed % 40,
   100→      apiUsage: 100 + (seed * 37) % 5000,
   101→      aiUsage: 1000 + (seed * 191) % 40000,
   102→      lastLogin: new Date(Date.now() - (seed % 14) * 86_400_000).toISOString(),
   103→    };
   104→  });
   105→}
   106→
   107→export function AdminUsersSection({ refreshKey }: { refreshKey: number }) {
   108→  const [users, setUsers] = React.useState<EnrichedUser[]>([]);
   109→  const [loading, setLoading] = React.useState(true);
   110→  const [totalUsers, setTotalUsers] = React.useState(0);
   111→  const [newThisMonth, setNewThisMonth] = React.useState(0);
   112→  const [adminCount, setAdminCount] = React.useState(0);
   113→  const [selected, setSelected] = React.useState<Set<string>>(new Set());
   114→
   115→  const [search, setSearch] = React.useState("");
   116→  const [roleFilter, setRoleFilter] = React.useState<"all" | "admin" | "user">("all");
   117→  const [planFilter, setPlanFilter] = React.useState<"all" | (typeof PLANS)[number]>("all");
   118→  const [sort, setSort] = React.useState<SortKey>("date");
   119→
   120→  const [addOpen, setAddOpen] = React.useState(false);
   121→  const [userToDelete, setUserToDelete] = React.useState<EnrichedUser | null>(null);
   122→  const [profileUser, setProfileUser] = React.useState<EnrichedUser | null>(null);
   123→  const [deleting, setDeleting] = React.useState(false);
   124→
   125→  const load = React.useCallback(async () => {
   126→    setLoading(true);
   127→    try {
   128→      const r = await fetch("/api/admin/stats").then((r) => r.json());
   129→      setTotalUsers(r.counts?.users ?? 0);
   130→      setNewThisMonth(r.counts?.usersThisMonth ?? 0);
   131→    } catch (e) {
   132→      console.error(e);
   133→    }
   134→    try {
   135→      const r = await fetch("/api/admin/users").then((r) => r.json() as Promise<{ users: AdminUser[] }>);
   136→      const enriched = enrichUsers(r.users ?? []);
   137→      setUsers(enriched);
   138→      setAdminCount(enriched.filter((u) => u.role === "admin").length);
   139→    } catch (e) {
   140→      console.error(e);
   141→      toast.error("Failed to load users");
   142→    } finally {
   143→      setLoading(false);
   144→    }
   145→  }, []);
   146→
   147→  React.useEffect(() => {
   148→    void load();
   149→  }, [load, refreshKey]);
   150→
   151→  async function patchUser(id: string, patch: { role?: string; plan?: string }) {
   152→    try {
   153→      const res = await fetch("/api/admin/users", {
   154→        method: "PATCH",
   155→        headers: { "Content-Type": "application/json" },
   156→        body: JSON.stringify({ id, ...patch }),
   157→      });
   158→      if (!res.ok) throw new Error("patch failed");
   159→      toast.success("User updated");
   160→      void load();
   161→    } catch (e) {
   162→      console.error(e);
   163→      toast.error("Failed to update user");
   164→    }
   165→  }
   166→
   167→  async function deleteUser() {
   168→    if (!userToDelete) return;
   169→    setDeleting(true);
   170→    try {
   171→      const res = await fetch(`/api/admin/users?id=${encodeURIComponent(userToDelete.id)}`, {
   172→        method: "DELETE",
   173→      });
   174→      if (!res.ok) throw new Error("delete failed");
   175→      toast.success(`Removed ${userToDelete.email}`);
   176→      setUserToDelete(null);
   177→      void load();
   178→    } catch (e) {
   179→      console.error(e);
   180→      toast.error("Failed to delete user");
   181→    } finally {
   182→      setDeleting(false);
   183→    }
   184→  }
   185→
   186→  // ---- Filter + sort ----
   187→  const filtered = React.useMemo(() => {
   188→    let list = users;
   189→    const q = search.trim().toLowerCase();
   190→    if (q) list = list.filter((u) => (u.name ?? "").toLowerCase().includes(q) || u.email.toLowerCase().includes(q));
   191→    if (roleFilter !== "all") list = list.filter((u) => u.role === roleFilter);
   192→    if (planFilter !== "all") list = list.filter((u) => u.plan === planFilter);
   193→    list = [...list].sort((a, b) => {
   194→      if (sort === "name") return (a.name ?? a.email).localeCompare(b.name ?? b.email);
   195→      if (sort === "plan") return (a.plan).localeCompare(b.plan);
   196→      return new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime();
   197→    });
   198→    return list;
   199→  }, [users, search, roleFilter, planFilter, sort]);
   200→
   201→  // ---- Bulk actions ----
   202→  function toggleOne(id: string) {
   203→    setSelected((prev) => {
   204→      const next = new Set(prev);
   205→      if (next.has(id)) next.delete(id);
   206→      else next.add(id);
   207→      return next;
   208→    });
   209→  }
   210→  function toggleAll() {
   211→    if (selected.size === filtered.length) setSelected(new Set());
   212→    else setSelected(new Set(filtered.map((u) => u.id)));
   213→  }

   221→          status: u.status, projects: u.projects, audits: u.audits,
   222→          apiUsage: u.apiUsage, aiUsage: u.aiUsage, createdAt: u.createdAt,
   223→        }));

   225→      toast.success(`Exported ${rows.length} users to CSV`);
   226→      return;
   227→    }
   228→    toast.success(`${action === "suspend" ? "Suspended" : "Activated"} ${selected.size} user(s)`, {
   229→      description: "Bulk action simulated in this preview build",

   231→    setSelected(new Set());
   232→  }

   247→            </Button>
   248→          </>

   251→

   253→        <StatCard label="Total Users" value={totalUsers} icon={Users} color="#10b981" />

   265→              placeholder="Search by name or email…"
   266→              value={search}
   267→              onChange={(e) => setSearch(e.target.value)}
   268→              className="pl-8"
   269→            />
   270→          </div>
   271→          <div className="flex flex-wrap items-center gap-2">
   272→            <Select value={roleFilter} onValueChange={(v) => setRoleFilter(v as typeof roleFilter)}>
   273→              <SelectTrigger className="w-[130px]">
   274→                <SelectValue placeholder="Role" />
   275→              </SelectTrigger>
   276→              <SelectContent>
   277→                <SelectItem value="all">All roles</SelectItem>
   278→                <SelectItem value="admin">Admin</SelectItem>
   279→                <SelectItem value="user">User</SelectItem>
   280→              </SelectContent>
   281→            </Select>
   282→            <Select value={planFilter} onValueChange={(v) => setPlanFilter(v as typeof planFilter)}>

   284→                <SelectValue placeholder="Plan" />
   285→              </SelectTrigger>
   286→              <SelectContent>
   287→                <SelectItem value="all">All plans</SelectItem>
   288→                {PLANS.map((p) => (
   289→                  <SelectItem key={p} value={p} className="capitalize">{p}</SelectItem>
   290→                ))}
   291→              </SelectContent>
   292→            </Select>
   293→            <Select value={sort} onValueChange={(v) => setSort(v as SortKey)}>

   295→                <SelectValue placeholder="Sort" />
   296→              </SelectTrigger>
   297→              <SelectContent>
   298→                <SelectItem value="date">Sort: Newest</SelectItem>
   299→                <SelectItem value="name">Sort: Name</SelectItem>

   301→              </SelectContent>
   302→            </Select>
   303→          </div>
   304→        </div>

   306→
   307→      {/* Bulk actions bar */}
   308→      {selected.size > 0 && (

   316→          </Button>
   317→          <Button size="sm" variant="outline" onClick={() => bulkAction("activate")}>
   318→            <CheckCircle2 className="w-3.5 h-3.5 mr-1" /> Activate
   319→          </Button>
   320→          <Button size="sm" variant="outline" onClick={() => bulkAction("export")}>
   321→            <Download className="w-3.5 h-3.5 mr-1" /> Export CSV
   322→          </Button>
   323→          <Button size="sm" variant="ghost" onClick={() => setSelected(new Set())}>

   325→          </Button>
   326→        </div>
   327→      )}
   328→

   330→      <Card className="p-5">

   332→          <Users className="w-4 h-4 text-emerald-600" /> Users
   333→          <Badge variant="secondary" className="ml-1">{filtered.length}</Badge>
   334→        </h3>
   335→        <div className={`-mx-2 px-2 ${SCROLLBAR_CLS} max-h-[60vh]`}>
   336→          {loading ? (
   337→            <SkeletonRows rows={6} cols={10} />
   338→          ) : filtered.length === 0 ? (
   339→            <EmptyState msg="No users match your filters" icon={Users} />
   340→          ) : (
   341→            <Table>
   342→              <TableHeader>
   343→                <TableRow>
   344→                  <TableHead className="w-10">
   345→                    <Checkbox
   346→                      checked={selected.size === filtered.length && filtered.length > 0}
   347→                      onCheckedChange={toggleAll}
   348→                    />
   349→                  </TableHead>
   350→                  <TableHead className="min-w-[180px]">User</TableHead>
   351→                  <TableHead>Role</TableHead>
   352→                  <TableHead>Plan</TableHead>
   353→                  <TableHead>Status</TableHead>
   354→                  <TableHead className="text-right">Projects</TableHead>
   355→                  <TableHead className="text-right">Audits</TableHead>
   356→                  <TableHead className="text-right">API Usage</TableHead>
   357→                  <TableHead className="text-right">AI Usage</TableHead>
   358→                  <TableHead className="whitespace-nowrap">Created</TableHead>
   359→                  <TableHead className="whitespace-nowrap">Last Login</TableHead>
   360→                  <TableHead className="text-right">Actions</TableHead>
   361→                </TableRow>
   362→              </TableHeader>
   363→              <TableBody className="zebra">
   364→                {filtered.map((u) => (
   365→                  <TableRow key={u.id} data-state={selected.has(u.id) ? "selected" : undefined}>
   366→                    <TableCell>
   367→                      <Checkbox
   368→                        checked={selected.has(u.id)}
   369→                        onCheckedChange={() => toggleOne(u.id)}
   370→                      />
   371→                    </TableCell>
   372→                    <TableCell>
   373→                      <div className="flex items-center gap-2">
   374→                        <Avatar className="w-8 h-8">
   375→                          <AvatarFallback className="bg-emerald-50 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-300 text-xs font-semibold">
   376→                            {initials(u.name, u.email)}
   377→                          </AvatarFallback>
   378→                        </Avatar>
   379→                        <div className="min-w-0">

   381→                          <div className="text-xs text-muted-foreground truncate">{u.email}</div>
   382→                        </div>
   383→                      </div>
   384→                    </TableCell>
   385→                    <TableCell>
   386→                      <Badge variant="outline" className={`capitalize ${roleBadgeClass(u.role)}`}>
   387→                        {u.role}
   388→                      </Badge>
   389→                    </TableCell>
   390→                    <TableCell>
   391→                      <Badge variant="outline" className={`capitalize ${planBadgeClass(u.plan)}`}>
   392→                        {u.plan}
   393→                      </Badge>
   394→                    </TableCell>
   395→                    <TableCell>
   396→                      {u.status === "active" ? (

   398→                          <span className="w-1.5 h-1.5 rounded-full bg-emerald-500" /> Active

   403→                        </span>
   404→                      )}
   405→                    </TableCell>
   406→                    <TableCell className="text-right tabular-nums">{u.projects}</TableCell>
   407→                    <TableCell className="text-right tabular-nums">{u.audits}</TableCell>
   408→                    <TableCell className="text-right tabular-nums text-xs">{formatCompact(u.apiUsage)}</TableCell>
   409→                    <TableCell className="text-right tabular-nums text-xs">{formatCompact(u.aiUsage)}</TableCell>
   410→                    <TableCell className="text-xs text-muted-foreground whitespace-nowrap">{formatDateLong(u.createdAt)}</TableCell>
   411→                    <TableCell className="text-xs text-muted-foreground whitespace-nowrap">{relativeTime(u.lastLogin)}</TableCell>

   424→                          </DropdownMenuItem>

   447→                                  onClick={() => patchUser(u.id, { plan: p })}
   448→                                  className="capitalize"
   449→                                >
   450→                                  {p}
   451→                                  {u.plan === p && (
   452→                                    <span className="ml-auto text-emerald-600">●</span>
   453→                                  )}

   455→                              ))}
   456→                            </DropdownMenuSubContent>
   457→                          </DropdownMenuSub>
   458→                          <DropdownMenuSub>
   459→                            <DropdownMenuSubTrigger className="flex items-center gap-2">
   460→                              <Shield className="w-3.5 h-3.5" /> Change role

   463→                              {ROLES.map((r) => (

   465→                                  key={r}
   466→                                  onClick={() => patchUser(u.id, { role: r })}
   467→                                  className="capitalize"
   468→                                >
   469→                                  {r}
   470→                                  {u.role === r && (
   471→                                    <span className="ml-auto text-emerald-600">●</span>
   472→                                  )}
   473→                                </DropdownMenuItem>
   474→                              ))}
   475→                            </DropdownMenuSubContent>
   476→                          </DropdownMenuSub>
   477→                          <DropdownMenuItem onClick={() => toast.success("Password reset email sent (demo)")}>
   478→                            <KeyRound className="w-3.5 h-3.5 mr-2" /> Reset password
   479→                          </DropdownMenuItem>
   480→                          <DropdownMenuSeparator />
   481→                          <DropdownMenuItem
   482→                            variant="destructive"
   483→                            onClick={() => setUserToDelete(u)}
   484→                            className="flex items-center gap-2"
   485→                          >
   486→                            <Trash2 className="w-3.5 h-3.5" /> Delete user
   487→                          </DropdownMenuItem>
   488→                        </DropdownMenuContent>
   489→                      </DropdownMenu>
   490→                    </TableCell>
   491→                  </TableRow>
   492→                ))}
   493→              </TableBody>

   498→
   499→      <AddUserDialog
   500→        open={addOpen}
   501→        onOpenChange={setAddOpen}

   505→      {/* Profile sheet */}

   511→                <SheetTitle className="flex items-center gap-3">
   512→                  <Avatar className="w-10 h-10">

   637→function UsageMeter({ label, value, max, color, format }: {
   638→  label: string; value: number; max: number; color: string; format?: boolean;
   639→}) {
   640→  const pct = Math.min(100, Math.round((value / max) * 100));
   641→  return (
   642→    <div className="mb-3 last:mb-0">
   643→      <div className="flex items-center justify-between text-xs mb-1">
   644→        <span className="text-muted-foreground">{label}</span>
   645→        <span className="tabular-nums font-medium">
   646→          {format ? formatCompact(value) : value} / {format ? formatCompact(max) : max}
   647→        </span>
   648→      </div>
   649→      <div className="h-1.5 w-full rounded-full bg-muted overflow-hidden">
   650→        <div
   651→          className="h-full rounded-full transition-all"
   652→          style={{ width: `${pct}%`, backgroundColor: color }}
   653→        />
   654→      </div>
   655→    </div>
   656→  );
   657→}

   659→function AddUserDialog({
   660→  open,
   661→  onOpenChange,
   662→  onCreated,
   663→}: {
   664→  open: boolean;
   665→  onOpenChange: (o: boolean) => void;
   666→  onCreated: () => void;
   667→}) {
   668→  const [email, setEmail] = React.useState("");
   669→  const [name, setName] = React.useState("");
   670→  const [role, setRole] = React.useState<string>("user");
   671→  const [plan, setPlan] = React.useState<string>("free");
   672→  const [saving, setSaving] = React.useState(false);
   673→
   674→  async function submit() {
   675→    if (!email.trim()) {
   676→      toast.error("Email is required");
   677→      return;
   678→    }
   679→    setSaving(true);
   680→    try {
   681→      const res = await fetch("/api/admin/users", {
   682→        method: "POST",
   683→        headers: { "Content-Type": "application/json" },
   684→        body: JSON.stringify({
   685→          email: email.trim(),
   686→          name: name.trim() || undefined,
   687→          role,
   688→          plan,
   689→        }),
   690→      });
   691→      if (!res.ok) {
   692→        const err = (await res.json().catch(() => ({}))) as { error?: string };
   693→        throw new Error(err.error ?? "create failed");
   694→      }
   695→      toast.success("User created");
   696→      setEmail("");
   697→      setName("");
   698→      setRole("user");
   699→      setPlan("free");
   700→      onOpenChange(false);
   701→      onCreated();
   702→    } catch (e) {
   703→      console.error(e);
   704→      toast.error(e instanceof Error ? e.message : "Failed to create user");
   705→    } finally {
   706→      setSaving(false);
   707→    }
   708→  }
   709→
   710→  return (
   711→    <Dialog open={open} onOpenChange={onOpenChange}>
   712→      <DialogContent className="sm:max-w-md">
   713→        <DialogHeader>
   714→          <DialogTitle>Add new user</DialogTitle>
   715→        </DialogHeader>
   716→        <div className="grid gap-3">
   717→          <div className="grid gap-1.5">
   718→            <Label htmlFor="au-email">Email</Label>
   719→            <Input
   720→              id="au-email"
   721→              type="email"
   722→              placeholder="user@example.com"
   723→              value={email}
   724→              onChange={(e) => setEmail(e.target.value)}
   725→            />
   726→          </div>
   727→          <div className="grid gap-1.5">
   728→            <Label htmlFor="au-name">Name (optional)</Label>
   729→            <Input
   730→              id="au-name"
   731→              placeholder="Jane Doe"
   732→              value={name}
   733→              onChange={(e) => setName(e.target.value)}
   734→            />
   735→          </div>
   736→          <div className="grid grid-cols-2 gap-3">
   737→            <div className="grid gap-1.5">
   738→              <Label>Role</Label>
   739→              <Select value={role} onValueChange={setRole}>
   740→                <SelectTrigger className="w-full">
   741→                  <SelectValue />
   742→                </SelectTrigger>
   743→                <SelectContent>
   744→                  {ROLES.map((r) => (
   745→                    <SelectItem key={r} value={r} className="capitalize">{r}</SelectItem>
   746→                  ))}
   747→                </SelectContent>
   748→              </Select>
   749→            </div>
   750→            <div className="grid gap-1.5">
   751→              <Label>Plan</Label>
   752→              <Select value={plan} onValueChange={setPlan}>
   753→                <SelectTrigger className="w-full">
   754→                  <SelectValue />
   755→                </SelectTrigger>
   756→                <SelectContent>
   757→                  {PLANS.map((p) => (
   758→                    <SelectItem key={p} value={p} className="capitalize">{p}</SelectItem>
   759→                  ))}
   760→                </SelectContent>
   761→              </Select>
   762→            </div>
   763→          </div>
   764→        </div>
   765→        <DialogFooter>
   766→          <DialogClose asChild>
   767→            <Button variant="outline" disabled={saving}>Cancel</Button>
   768→          </DialogClose>
   769→          <Button onClick={submit} disabled={saving} className={EMERALD_BTN}>
   770→            {saving && <Loader2 className="w-3.5 h-3.5 mr-1 animate-spin" />}
   771→            Create user
   772→          </Button>
   773→        </DialogFooter>
   774→      </DialogContent>
   775→    </Dialog>
   776→  );
   777→}
