   111→   111→function planBadge(plan: string): string {
   112→   112→  const map: Record<string, string> = {
   113→   113→    free: "bg-slate-100 text-slate-700 border-slate-200",
   114→   114→    starter: "bg-sky-50 text-sky-700 border-sky-200",
   115→   115→    pro: "bg-emerald-50 text-emerald-700 border-emerald-200",
   116→   116→    agency: "bg-violet-50 text-violet-700 border-violet-200",
   117→   117→  };
   118→   118→  return map[plan] ?? "bg-slate-100 text-slate-700 border-slate-200";
   119→   119→}

   121→   121→function roleBadge(role: string): string {

   127→   127→function scoreColor(v: number): string {

   131→   131→function formatDate(iso: string): string {
   132→   132→  if (!iso) return "—";
   133→   133→  const d = new Date(iso);
   134→   134→  if (isNaN(d.getTime())) return "—";
   135→   135→  return d.toLocaleString(undefined, {
   136→   136→    month: "short",
   137→   137→    day: "numeric",
   138→   138→    hour: "2-digit",
   139→   139→    minute: "2-digit",
   140→   140→  });
   141→   141→}
   142→   142→
   143→   143→function formatDateLong(iso: string): string {
   144→   144→  if (!iso) return "—";
   145→   145→  const d = new Date(iso);
   146→   146→  if (isNaN(d.getTime())) return "—";
   147→   147→  return d.toLocaleDateString(undefined, {
   148→   148→    year: "numeric",
   149→   149→    month: "short",
   150→   150→    day: "numeric",
   151→   151→  });
   152→   152→}
   153→   153→
   154→   154→function initials(name: string | null, email: string): string {
   155→   155→  if (name) {
   156→   156→    const parts = name.trim().split(/\s+/);
   157→   157→    return (parts[0]?.[0] ?? "") + (parts[1]?.[0] ?? parts[0]?.[1] ?? "");
   158→   158→  }
   159→   159→  return email.slice(0, 2);
   160→   160→}

   612→   612→function AddUserDialog({
   613→   613→  open,
   614→   614→  onOpenChange,
   615→   615→  onCreated,
   616→   616→}: {
   617→   617→  open: boolean;
   618→   618→  onOpenChange: (o: boolean) => void;
   619→   619→  onCreated: () => void;
   620→   620→}) {
   621→   621→  const [email, setEmail] = React.useState("");
   622→   622→  const [name, setName] = React.useState("");
   623→   623→  const [role, setRole] = React.useState<string>("user");
   624→   624→  const [plan, setPlan] = React.useState<string>("free");
   625→   625→  const [saving, setSaving] = React.useState(false);
   626→   626→
   627→   627→  async function submit() {
   628→   628→    if (!email.trim()) {
   629→   629→      toast.error("Email is required");
   630→   630→      return;
   631→   631→    }
   632→   632→    setSaving(true);
   633→   633→    try {
   634→   634→      const res = await fetch("/api/admin/users", {
   635→   635→        method: "POST",
   636→   636→        headers: { "Content-Type": "application/json" },
   637→   637→        body: JSON.stringify({
   638→   638→          email: email.trim(),
   639→   639→          name: name.trim() || undefined,
   640→   640→          role,
   641→   641→          plan,
   642→   642→        }),
   643→   643→      });
   644→   644→      if (!res.ok) {
   645→   645→        const err = (await res.json().catch(() => ({}))) as { error?: string };
   646→   646→        throw new Error(err.error ?? "create failed");
   647→   647→      }
   648→   648→      toast.success("User created");
   649→   649→      setEmail("");
   650→   650→      setName("");
   651→   651→      setRole("user");
   652→   652→      setPlan("free");
   653→   653→      onOpenChange(false);
   654→   654→      onCreated();
   655→   655→    } catch (e) {
   656→   656→      console.error(e);
   657→   657→      toast.error(e instanceof Error ? e.message : "Failed to create user");
   658→   658→    } finally {
   659→   659→      setSaving(false);
   660→   660→    }
   661→   661→  }
   662→   662→
   663→   663→  return (
   664→   664→    <Dialog open={open} onOpenChange={onOpenChange}>
   665→   665→      <DialogContent className="sm:max-w-md">

   672→   672→            <Input
   673→   673→              id="au-email"
   674→   674→              type="email"
   675→   675→              placeholder="user@example.com"
   676→   676→              value={email}
   677→   677→              onChange={(e) => setEmail(e.target.value)}
   678→   678→            />
   679→   679→          </div>
   680→   680→          <div className="grid gap-1.5">
   681→   681→            <Label htmlFor="au-name">Name (optional)</Label>
   682→   682→            <Input
   683→   683→              id="au-name"
   684→   684→              placeholder="Jane Doe"
   685→   685→              value={name}
   686→   686→              onChange={(e) => setName(e.target.value)}
   687→   687→            />
   688→   688→          </div>
   689→   689→          <div className="grid grid-cols-2 gap-3">
   690→   690→            <div className="grid gap-1.5">
   691→   691→              <Label>Role</Label>
   692→   692→              <Select value={role} onValueChange={setRole}>
   693→   693→                <SelectTrigger className="w-full">
   694→   694→                  <SelectValue />
   695→   695→                </SelectTrigger>
   696→   696→                <SelectContent>
   697→   697→                  {ROLES.map((r) => (
   698→   698→                    <SelectItem key={r} value={r} className="capitalize">
   699→   699→                      {r}
   700→   700→                    </SelectItem>
   701→   701→                  ))}
   702→   702→                </SelectContent>
   703→   703→              </Select>
   704→   704→            </div>
   705→   705→            <div className="grid gap-1.5">
   706→   706→              <Label>Plan</Label>
   707→   707→              <Select value={plan} onValueChange={setPlan}>
   708→   708→                <SelectTrigger className="w-full">
   709→   709→                  <SelectValue />
   710→   710→                </SelectTrigger>
