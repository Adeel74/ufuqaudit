   100→function truncateUrl(u: string, n = 28): string {
   101→  const stripped = u.replace(/^https?:\/\//, "");
   102→  if (stripped.length <= n) return stripped;
   103→  return stripped.slice(0, n - 1) + "…";
   104→}
   105→
   106→function relTime(iso: string | null): string {
   107→  if (!iso) return "Never";
   108→  const t = new Date(iso).getTime();
   109→  if (Number.isNaN(t)) return "Never";
   110→  const diff = Date.now() - t;
   111→  const fmt = (abs: number, suffix: string) => {
   112→    if (abs < 60_000) return `<1m${suffix}`;
   113→    const mins = Math.floor(abs / 60_000);
   114→    if (mins < 60) return `${mins}m${suffix}`;
   115→    const hours = Math.floor(mins / 60);
   116→    if (hours < 24) return `${hours}h${suffix}`;
   117→    const days = Math.floor(hours / 24);
   118→    if (days < 30) return `${days}d${suffix}`;
   119→    const months = Math.floor(days / 30);
   120→    return `${months}mo${suffix}`;
   121→  };
   122→  if (diff < 0) return fmt(-diff, "");
   123→  return fmt(diff, " ago");
   124→}
   125→
   126→function isExpired(link: PortalLink): boolean {
   127→  if (!link.expiresAt) return false;
   128→  return new Date(link.expiresAt).getTime() < Date.now();
   129→}
   130→
   131→function portalPath(token: string): string {
   132→  return `/portal/${token}`;
   133→}
   134→
   135→function buildFullUrl(token: string): string {
   136→  if (typeof window === "undefined") return portalPath(token);
   137→  return `${window.location.origin}${portalPath(token)}`;
   138→}

   569→        open={dialogOpen}
   570→        onOpenChange={setDialogOpen}
   571→        form={form}
   572→        setForm={setForm}
   573→        onCreate={onCreate}
   574→        adding={adding}
   575→      />
   576→
   577→      {/* Revoke confirmation */}
   578→      <AlertDialog
   579→        open={revokeTarget !== null}
   580→        onOpenChange={(o) => !o && !revoking && setRevokeTarget(null)}
   581→      >
   582→        <AlertDialogContent>
   583→          <AlertDialogHeader>
   584→            <AlertDialogTitle>Revoke portal link?</AlertDialogTitle>
   585→            <AlertDialogDescription>
   586→              The link shared with{" "}
   587→              <span className="font-medium text-foreground">{revokeTarget?.clientName}</span>{" "}
   588→              will stop working immediately. The client will no longer be able to access the audit report.
   589→            </AlertDialogDescription>
   590→          </AlertDialogHeader>
   591→          <AlertDialogFooter>
   592→            <AlertDialogCancel disabled={revoking}>Keep link</AlertDialogCancel>
   593→            <AlertDialogAction
   594→              onClick={(e) => {
   595→                e.preventDefault();
   596→                onConfirmRevoke();
   597→              }}
   598→              disabled={revoking}
   599→              className="bg-red-600 hover:bg-red-700 text-white"
   600→            >
   601→              {revoking ? (
   602→                <>
   603→                  <Loader2 className="w-3.5 h-3.5 mr-1 animate-spin" /> Revoking…
   604→                </>
   605→              ) : (
   606→                <>
   607→                  <Ban className="w-3.5 h-3.5 mr-1" /> Revoke link
   608→                </>
   609→              )}
   610→            </AlertDialogAction>
   611→          </AlertDialogFooter>
   612→        </AlertDialogContent>
   613→      </AlertDialog>
   614→    </div>
   615→  );
