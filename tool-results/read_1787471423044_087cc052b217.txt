   118→function truncate(s: string, n: number): string {
   119→  if (s.length <= n) return s;
   120→  return s.slice(0, n - 1) + "…";
   121→}

   123→function relativeTime(iso: string): string {
   124→  const then = new Date(iso).getTime();
   125→  const now = Date.now();
   126→  const diffMs = Math.max(0, now - then);
   127→  const mins = Math.floor(diffMs / 60_000);
   128→  const hours = Math.floor(diffMs / 3_600_000);
   129→  const days = Math.floor(diffMs / 86_400_000);
   130→  if (mins < 1) return "just now";
   131→  if (mins < 60) return `${mins}m ago`;
   132→  if (hours < 24) return `${hours}h ago`;
   133→  if (days < 30) return `${days}d ago`;
   134→  const months = Math.floor(days / 30);
   135→  if (months < 12) return `${months}mo ago`;
   136→  return `${Math.floor(months / 12)}y ago`;
   137→}
